🏗️ MODELS app/Models/Post.php
namespace App\Models;
use Illuminate\Database\Eloquent\Model;

class Post extends Model {

  // Auto maps: Post → posts table
  // (Laravel pluralizes class name)

  // ✅ WHITELIST (recommended)
  protected $fillable = [
    'title', 'description', 'author_id',
  ];

  // OR: BLACKLIST approach
  protected $guarded = ['is_admin'];
}

// ── Without $fillable ──
// Post::create([...]) throws:
// MassAssignmentException!

// ── Safe alternative (no $fillable) ──
$post = new Post();
$post->title = 'Hello World';
$post->save();
🔍 DB FACADE — READ
use Illuminate\Support\Facades\DB;

// Find by PK → stdClass|null
$post = DB::table('posts')->find(20);

// Get ALL → Collection<stdClass>
$posts = DB::table('posts')->get();

// WHERE + first match
$post = DB::table('posts')
           ->where('slug', 'first-post')
           ->first();

// Very first row
$first = DB::table('posts')->first();

// Chain wheres
$posts = DB::table('posts')
           ->where('author_id', 5)
           ->where('status', 'published')
           ->get();

// ORDER + LIMIT
$posts = DB::table('posts')
   ->orderBy('created_at', 'desc')
   ->limit(10)->get();
Write Operations & Relationships
✏️ DB FACADE — CREATE / UPDATE / DELETE
// ── CREATE → returns bool ──
DB::table('posts')->insert([
  'title' => 'First Post',
  'desc'  => 'Some description',
]);

// ── UPDATE → returns affected rows ──
DB::table('posts')
  ->where('id', 1)
  ->update(['title' => 'New Title']);

// Update multiple columns:
DB::table('posts')
  ->where('id', 1)
  ->update([
    'title' => 'New Title',
    'desc'  => 'New Desc',
  ]);

// ── DELETE → returns affected rows ──
DB::table('posts')
  ->where('id', 1)
  ->delete();

// Delete with condition:
DB::table('posts')
  ->where('votes', '<', 5)
  ->delete();

use App\Models\Post;

// ── DELETE ──

Post::find(25)->delete();

🔗 ELOQUENT RELATIONSHIPS
// ── 1-to-Many ──

// User Model (the "1" side):
class User extends Model {
  public function posts() {
    return $this->hasMany(
      Post::class, 'author_id'
    );
  }
}

// Post Model (the "Many" side):
class Post extends Model {
  public function user() {
    return $this->belongsTo(
      User::class, 'author_id'
    );
  }
  public function sections() {
    return $this->hasMany(Section::class);
  }
}

// ── Using Relations in Controller ──
$user->posts;           // Collection<Post>
$post->user;            // User object
Post::find(23)->sections;// Collection

// ── Eager Load (N+1 fix) ──
Post::with('user')->get();
Post::with(['user', 'sections'])->get();
🏭 DATABASE FACTORY database/factories/
// php artisan make:factory PostFactory

class PostFactory extends Factory {
  public function definition(): array {
    return [
      'title'       => $this->faker->sentence(),
      'description' => $this->faker->paragraph(),
      'author_id'   => $this->faker
                          ->numberBetween(1, 100),
      'created_at'  => now(),
    ];
  }
}

// ── Faker Helpers ──
$this->faker->sentence()       // "Lorem ipsum..."
$this->faker->paragraph()      // multi-sentence
$this->faker->name()           // "John Doe"
$this->faker->email()          // "john@example.com"
$this->faker->url()
$this->faker->imageUrl(640, 480)
$this->faker->randomElement(['a','b'])

// ── Usage ──
Post::factory()->create();
Post::factory()->count(500)->create();

📥 HTTP REQUEST HANDLING
use Illuminate\Http\Request;

public function store(Request $request) {

  // all() → ALL inputs as array
  // ⚠️ Needs $fillable in Model!
  Post::create($request->all());

  // only() → WHITELIST specific keys
  Post::create($request->only([
    'title', 'description', 'author_id',
  ]));

  // except() → BLACKLIST specific keys
  Post::create($request->except(['_token']));

  // Single value:
  $title = $request->input('title');
  $title = $request->title; // magic prop

  // Check existence:
  if ($request->has('title')) { /* ... */ }

  // Redirect after save:
  return redirect()
    ->route('posts.index')
    ->with('success', 'Saved!');
}
⚖️ DB FACADE vs ELOQUENT
Operation DB Facade Eloquent
All rows DB::table('posts')->get() Post::all()
By ID ->find(20) Post::find(20)
Where ->where('k','v')->first() Post::where('k','v')->first()
Insert ->insert([...]) Post::create([...])
Update ->where()->update([...]) Post::find(n)->update([...])
Delete ->where()->delete() Post::where()->delete()
Returns stdClass / bool / int Model object / Collection
Relations ❌ Manual joins ✅ hasMany, belongsTo

// ── Migration for comments table ──
$t->id();

$t->morphs('commentable');
// → adds commentable_id + commentable_type
$t->timestamps();

// ── Comment Model ──
class Comment extends Model {
  public function commentable() {
    return $this->morphTo();
  }
}

// ── Post Model ──
class Post extends Model {
  public function comments() {
    return $this->morphMany(
      Comment::class, 'commentable'
    );
  }
}

🗓️ CARBON DATES + PAGINATION
// ── Carbon Date Formatting ──
// Carbon is auto-cast on Eloquent timestamps

$post->created_at->format('d M Y');
// → "15 May 2025"

$post->created_at->format('d/m/Y H:i');
// → "15/05/2025 14:30"

$post->created_at->diffForHumans();
// → "3 days ago"

$post->created_at->toDateString();
// → "2025-05-15"

// In Blade:
// {{ $post->created_at->format('d M Y') }}

// ── Pagination ──
// Controller:
$posts = Post::paginate(15);
// or with query:
$posts = Post::with('user')
    ->orderBy('created_at', 'desc')
    ->paginate(15);

// Blade view:
{{ $posts->links() }}
// → renders Tailwind pagination UI
